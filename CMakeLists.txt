cmake_minimum_required(VERSION 3.25)
project(silent-runner VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "SilentRunner requires Windows x64 and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /permissive- /utf-8)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_library(runner_core src/cli/arguments.cpp src/process/launcher.cpp src/windows/error.cpp)
target_include_directories(runner_core PUBLIC src)
add_executable(SilentRunner WIN32 src/main.cpp src/SilentRunner.rc)
target_link_libraries(SilentRunner PRIVATE runner_core shell32)
target_compile_definitions(SilentRunner PRIVATE RUNNER_VERSION=L"${PROJECT_VERSION}")
include(CTest)
if(BUILD_TESTING)
  add_executable(runner_probe tests/probe.cpp)
  add_executable(runner_tests tests/tests.cpp)
  target_link_libraries(runner_tests PRIVATE runner_core advapi32)
  add_test(NAME runner_tests COMMAND runner_tests $<TARGET_FILE:runner_probe> $<TARGET_FILE:SilentRunner>)
  set_tests_properties(runner_tests PROPERTIES TIMEOUT 60)
  find_program(WINDOWS_POWERSHELL powershell REQUIRED)
  find_program(POWERSHELL_CORE pwsh)
  add_test(NAME windows_powershell_and_cmd COMMAND "${WINDOWS_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/shells.ps1" -Runner $<TARGET_FILE:SilentRunner>)
  if(POWERSHELL_CORE)
    add_test(NAME powershell_7_and_cmd COMMAND "${POWERSHELL_CORE}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/shells.ps1" -Runner $<TARGET_FILE:SilentRunner>)
  endif()
endif()
install(TARGETS SilentRunner RUNTIME DESTINATION .)
install(FILES README.md DESTINATION .)
install(FILES docs/usage.md docs/validation.md DESTINATION docs)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "SilentRunner-${PROJECT_VERSION}-windows-x64")
include(CPack)
